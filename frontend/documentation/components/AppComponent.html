import { Component } from '@angular/core';
import { CommonModule } from '@angular/common';
import { RouterOutlet } from '@angular/router';
import { AuthService } from './services/auth.service';

@Component({
    selector: 'app-root',
    standalone: true,
    imports: [CommonModule, RouterOutlet],
    template: `
    <nav class="navbar" *ngIf="authService.isAuthenticated()">
      <div class="nav-brand">🏥 Clínica Médica</div>
      <div class="nav-user">
        <span><b>{{ usuario?.nombres }} {{ usuario?.apellidos }}</b> ({{ usuario?.rol }})</span>
        <button class="btn-logout" (click)="authService.logout()">Cerrar Sesión</button>
      </div>
    </nav>
    <main class="container">
      <router-outlet></router-outlet>
    </main>
  `,
    styles: [`
    .navbar {
      display: flex;
      justify-content: space-between;
      align-items: center;
      background: #1e293b;
      color: white;
      padding: 0.8rem 1.5rem;
      font-family: Arial, sans-serif;
    }
    .nav-brand { font-size: 1.2rem; font-weight: bold; }
    .nav-user { display: flex; align-items: center; gap: 1rem; font-size: 0.9rem; }
    .btn-logout {
      background: #ef4444;
      color: white;
      border: none;
      padding: 0.4rem 0.8rem;
      border-radius: 4px;
      cursor: pointer;
      font-weight: bold;
    }
    .btn-logout:hover { background: #dc2626; }
    .container { padding: 1.5rem; max-width: 1000px; margin: 0 auto; font-family: Arial, sans-serif; }
  `],
})
export class AppComponent {
    constructor(public authService: AuthService) { }

    get usuario() {
        return this.authService.getUsuario();
    }
}
